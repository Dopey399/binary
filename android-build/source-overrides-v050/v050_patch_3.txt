     state.x += (best.x - state.x) * follow;

+        } catch (_) {}
+      }

+    return { x: state.x, y: state.y, radius: Math.min(g.w, g.h) * (Core.clamp(Number(effects.targetSize) || 12, 2, 40) / 100) };
